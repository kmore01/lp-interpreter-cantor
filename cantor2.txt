grammar cantor;

prog: statement* EOF;

statement: directive | function_def;

directive
    : 'main' ID
    | 'import' ID
    | 'extended'
    ;

function_def: 'define' ID DOC? operation;

operation
    : 'comp' ID ID        # CompOp
    | 'pair' ID ID        # PairOp
    | 'compair' ID ID ID  # CompairOp
    | 'mu' ID             # MuOp
    | 'primrec' ID ID ID  # PrimrecOp
    | ID                  # AliasOp
    ;

DOC : '[' ~[\r\n]* ']' ;
COMMENT : '#' ~[\r\n]* -> skip ;
ID      : [a-zA-Z_][a-zA-Z0-9_]* ;
WS      : [ \t\r\n]+    -> skip ;
